import { router, lazy } from 'nav-router';

export const app = router({
  title: 'Acme',
  layout: RootLayout,
  index: { path: '/', view: HomePage },

  routes: {
    team: {
      path: '/team/:org',
!!      params: { org: { allowed: ['acme', 'globex'] } },
      layout: TeamLayout,
      data: ({ params, signal }) => ({
        team: api.team(params.org, { signal }),
      }),
      routes: {
        members: {
          path: '/team/:org/members',
!!          data: ({ data, signal }) => api.members(data.team.id, { signal }),
!!          view: lazy(() => import('./members.js')),
        },
        settings: {
          path: '/team/:org/settings',
!!          permission: ({ user }) => user?.role === 'admin',
          view: SettingsPage,
        },
      },
    },
  },
});

// the same app, on the Navigation API alone
const PAGES = {
  home: new URLPattern({ pathname: '/' }),
  members: new URLPattern({ pathname: '/team/:org/members' }),
  settings: new URLPattern({ pathname: '/team/:org/settings' }),
};
const teams = new Map();
let mountedTeam = null;

navigation.addEventListener('navigate', (event) => {
  // downloads, form posts, other sites and #jumps are the browser's
  if (!event.canIntercept || event.downloadRequest !== null) return;
  if (event.formData || event.hashChange) return;

  const url = new URL(event.destination.url);
  const [page, match] = Object.entries(PAGES)
    .map(([name, pattern]) => [name, pattern.exec(url)])
    .find(([, found]) => found) ?? [];
  if (!page) return;
  const { org } = match.pathname.groups;
  if (page !== 'home' && !['acme', 'globex'].includes(org)) return;

  // refuse before the URL moves, or the page lands in history
  if (page === 'settings' && currentUser()?.role !== 'admin') {
    event.preventDefault();
    const returnTo = encodeURIComponent(url.pathname);
    navigation.navigate(`/sign-in?returnTo=${returnTo}`);
    return;
  }

  event.intercept({
    scroll: 'after-transition',
    focusReset: 'after-transition',
    async handler() {
      const { signal } = event;
      document.title = 'Acme';
      if (page === 'home') {
        mountedTeam = null;
        return render('#app', RootLayout(HomePage()));
      }
      // the team's data loads once, and its layout stays mounted
      if (!teams.has(org)) {
        teams.set(org, await api.team(org, { signal }));
      }
      if (signal.aborted) return; // a newer navigation won
      if (mountedTeam !== org) {
        render('#app', RootLayout(TeamLayout(teams.get(org))));
        mountedTeam = org;
      }
      // a page's data needs what its section loaded, so it waits for it
      const team = teams.get(org);
      const members = page === 'members'
        ? await api.members(team.id, { signal })
        : undefined;
      const view = page === 'members'
        ? (await import('./members.js')).MembersPage
        : SettingsPage;
      if (signal.aborted) return;
      render('#team-page', view({ team, members }));
    },
  });
});

{{t:home.xray.note}}

Built on what the platform already does

No history shim, no click interception, no <Link> component. Plain anchors and forms are navigations, because the browser says so.

Every navigation can be cancelled

Each route's data gets the navigation's AbortSignal. Click away mid-fetch and the request stops.

data: ({ params, signal }) =>
  api.team(params.org, { signal })

Back and forward keep their data

Results ride on the history entry. Going back shows what was there, without fetching it again.

navigation.back()
// restored from the entry, no refetch

Guards run before the URL moves

A redirect goes straight to its target. A page the user can't open never enters history.

permission: ({ user }) => !!user,
signInRoute: 'signIn',
